CREATE TABLE "users" ("id" integer primary key autoincrement not null, "name" varchar not null, "email" varchar not null, "email_verified_at" datetime, "password" varchar not null, "remember_token" varchar, "created_at" datetime, "updated_at" datetime, "role" varchar check ("role" in ('admin', 'driver', 'customer')) not null default 'customer', "status" tinyint(1) not null default '1', "phone" varchar, "profile_photo" varchar)
